cmake_minimum_required(VERSION 3.22)
project(sms_mobile_probe LANGUAGES CXX)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI MATCHES "^(arm64-v8a|x86_64)$")
  message(FATAL_ERROR "This diagnostic requires the Android NDK and a supported 64-bit ABI")
endif()
add_executable(sms_mobile_probe memory_probe.cpp)
target_compile_features(sms_mobile_probe PRIVATE cxx_std_17)
target_compile_options(sms_mobile_probe PRIVATE -Wall -Wextra -Werror)
add_library(sms_mobile_probe_jni SHARED memory_probe.cpp)
target_compile_features(sms_mobile_probe_jni PRIVATE cxx_std_17)
target_compile_options(sms_mobile_probe_jni PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(sms_mobile_probe_jni PRIVATE SMS_PROBE_JNI=1)
foreach(target sms_mobile_probe sms_mobile_probe_jni)
  target_sources(${target} PRIVATE
    ../../platform/mobile/spc_native_registry.cpp spc_registry_test.cpp)
  target_compile_definitions(${target} PRIVATE SMS_SPC_REGISTRY_TEST_NO_MAIN=1)
endforeach()
add_executable(sms_gx_static_test gx_static_memory_test.cpp
    ../../platform/mobile/gx_static_memory.cpp)
target_compile_features(sms_gx_static_test PRIVATE cxx_std_17)
target_compile_options(sms_gx_static_test PRIVATE -Wall -Wextra -Werror)
